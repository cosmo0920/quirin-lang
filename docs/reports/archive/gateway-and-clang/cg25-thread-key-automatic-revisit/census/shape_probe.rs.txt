const THREADS: &str =
    include_str!("../../../providers/quirin-provider-libc/contracts/libc/threads.qv");
const REGEX: &str = include_str!("../../../providers/quirin-provider-libc/contracts/libc/regex.qv");
const STDLIB: &str =
    include_str!("../../../providers/quirin-provider-libc/contracts/libc/memory.qv");

const DECLARATIONS: &str = "\
typedef unsigned int pthread_key_t;
typedef struct { void *buffer; unsigned long re_nsub; } regex_t;
extern int pthread_key_create (pthread_key_t *__key, void (*__destr_function) (void *));
extern int pthread_key_delete (pthread_key_t __key);
extern void *pthread_getspecific (pthread_key_t __key);
extern int pthread_setspecific (pthread_key_t __key, const void *__pointer);
typedef int pthread_once_t;
extern int pthread_once (pthread_once_t *__once_control, void (*__init_routine) (void));
int regcomp(regex_t *preg, const char *pattern, int cflags);
int regexec(const regex_t *preg, const char *string, unsigned long nmatch, void *pmatch, int eflags);
void regfree(regex_t *preg);
void free(void *p);
void *malloc(unsigned long n);
void use_key(pthread_key_t k);
struct holder { pthread_key_t k; };
pthread_key_t global_key;
regex_t global_re;
";

fn contracts() -> ResourceContracts {
    load_resource_contract_sources(&BTreeMap::from([
        ("libc/threads.qv".to_owned(), THREADS.to_owned()),
        ("libc/regex.qv".to_owned(), REGEX.to_owned()),
        ("libc/memory.qv".to_owned(), STDLIB.to_owned()),
    ]))
    .expect("load")
}

fn show(label: &str, source: &str) {
    let text = format!("{DECLARATIONS}\n{source}\n");
    let outcome = analyze_unrestricted_c11(
        &C11TranslationUnit {
            source: &text,
            ..Default::default()
        },
        "subject",
        &contracts(),
    );
    let outcome = match outcome {
        Ok(outcome) => outcome,
        Err(error) => {
            println!("PROBE {label}: CLANG-REJECTED {error:?}");
            return;
        }
    };
    let summary = match &outcome {
        UnrestrictedLegacyResult::Safe(certificate) => Some(&certificate.summary),
        UnrestrictedLegacyResult::Unsafe(summary) => Some(summary),
        UnrestrictedLegacyResult::Unknown { partial, .. } => partial.as_ref(),
    };
    let Some(result) = summary.and_then(|summary| summary.functions.get("subject")) else {
        println!("PROBE {label}: NOT PROJECTED {outcome:?}");
        return;
    };
    let kinds = result
        .diagnostics
        .iter()
        .map(|d| format!("L{} {:?}", d.line, d.kind))
        .collect::<Vec<_>>();
    println!("PROBE {label}: {:?} {kinds:?}", result.status);
}
