//! source and output are archived under docs/reports/w45-sr5-soundness/probes/.
//! Nothing asserts: every row prints its status and diagnostics.

use std::collections::BTreeMap;

use qcore_c_gateway::legacy_c::{
    C11TranslationUnit, CCompilerDialect, LegacyDiagnosticKind, LegacyFunctionScope,
    LegacyProgramSummary, ResourceContracts, UnrestrictedLegacyResult,
    analyze_legacy_c_translation_unit, analyze_unrestricted_c11,
};
use qcore_contract::contract_loader::load_resource_contract_sources;

fn load(extra: &[(&str, &str)]) -> ResourceContracts {
    let mut sources = BTreeMap::from([
        (
            "q_base.qcore".to_owned(),
            include_str!("fixtures/contracts/q_base.qcore").to_owned(),
        ),
        (
            "q_nullable.qcore".to_owned(),
            include_str!("fixtures/contracts/q_nullable.qcore").to_owned(),
        ),
    ]);
    for (name, text) in extra {
        sources.insert((*name).to_owned(), (*text).to_owned());
    }
    load_resource_contract_sources(&sources).expect("contracts load")
}

fn kinds(summary: &LegacyProgramSummary, function: &str) -> Vec<String> {
    summary
        .functions
        .get(function)
        .map(|f| {
            f.diagnostics
                .iter()
                .map(|d| match &d.kind {
                    LegacyDiagnosticKind::Unsupported { description } => {
                        format!("Unsupported({description})")
                    }
                    other => format!("{other:?}").chars().take(160).collect(),
                })
                .collect()
        })
        .unwrap_or_default()
}

fn text_rows(label: &str, source: &str, entry: &str, functions: &[&str], contracts: &ResourceContracts) {
    match analyze_legacy_c_translation_unit(source, entry, contracts) {
        Ok(summary) => {
            for function in functions {
                let status = summary.functions.get(*function).map(|f| f.status);
                let records = summary
                    .functions
                    .get(*function)
                    .map(|f| {
                        f.counted_users
                            .iter()
                            .map(|r| format!("{}={:?}", r.clause, r.verdict))
                            .collect::<Vec<_>>()
                    })
                    .unwrap_or_default();
                println!(
                    "PROBE {label} [{function}]: status={status:?} diagnostics={:?}{}",
                    kinds(&summary, function),
                    if records.is_empty() { String::new() } else { format!(" clauses={records:?}") }
                );
            }
        }
        Err(error) => println!("PROBE {label}: tokenize error {error:?}"),
    }
}

fn c11_row(label: &str, source: &str, scope: LegacyFunctionScope, contracts: &ResourceContracts) {
    let outcome = analyze_unrestricted_c11(
        &C11TranslationUnit {
            source,
            dialect: CCompilerDialect::Gnu17,
            function_scope: scope,
            ..Default::default()
        },
        "subject",
        contracts,
    );
    match outcome {
        Ok(result) => {
            let verdict = result.verdict();
            let summary = match &result {
                UnrestrictedLegacyResult::Safe(certificate) => Some(&certificate.summary),
                UnrestrictedLegacyResult::Unsafe(summary) => Some(summary),
                UnrestrictedLegacyResult::Unknown { partial, .. } => partial.as_ref(),
            };
            let reason = match &result {
                UnrestrictedLegacyResult::Unknown { reason, .. } => reason.clone(),
                _ => String::new(),
            };
            let rows = summary
                .map(|s| {
                    s.functions
                        .iter()
                        .map(|(name, f)| format!("{name}={:?}{:?}", f.status, kinds(s, name)))
                        .collect::<Vec<_>>()
                })
                .unwrap_or_default();
            println!("PROBE {label}: unit_verdict={verdict:?} reason={reason:?} functions={rows:?}");
        }
        Err(error) => println!("PROBE {label}: refused {error:?}"),
    }
}

    // ---- 1. Rule 5k: a member release and a callee that releases it. ----

        text_rows(

    k("5k-tested-member-then-destroy", DESTROY, "if (n->buf) q_nullable_free(n->buf); node_destroy(n);", &plain);
    k("5k-member-then-wrapper", &format!("{DESTROY}{DROP}"), "q_nullable_free(n->buf); node_drop(n);", &plain);
    k("5k-destroy-then-member", DESTROY, "node_destroy(n); q_nullable_free(n->buf); q_nullable_free(n);", &plain);
    k("5k-caller-alias", DESTROY, "void *b = n->buf; q_nullable_free(b); node_destroy(n);", &plain);
    k("5k-callee-alias", DESTROY_ALIAS, "q_nullable_free(n->buf); node_destroy_alias(n);", &plain);
    k("5k-callee-alias-control", DESTROY_ALIAS, "node_destroy_alias(n);", &plain);
    k("5k-other-member", DESTROY_NEXT, "q_nullable_free(n->buf); node_destroy_next(n);", &plain);
    k("5k-same-member-next", DESTROY_NEXT, "q_nullable_free(n->next); node_destroy_next(n);", &plain);
    let declared = load(&[("destroy.qcore", DECLARED_DESTROY)]);
    k("5k-declared-defined-destroy", DESTROY, "q_nullable_free(n->buf); node_destroy(n);", &declared);
    k(
        "5k-declared-undefined-destroy",
        "void node_destroy(struct node *n);\n",
        "q_nullable_free(n->buf); node_destroy(n);",
        &declared,
    );
    // ---- 2. The held prefixes, spelled by the source and minted by the tool. ----

            "k_close",
            "k.qcore".into(),

            Some("contracts.lock".into()),
            "probe".into(),

            label,

    h("held-review-prefix", "qcore_review_pending__zz(0);", &plain);

    text_rows(
        "held-prefix-defined",

    // ---- 4. 10f's static root, shadowed. ----
    c(
